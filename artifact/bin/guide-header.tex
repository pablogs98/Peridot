% Long commands and URLs in code blocks wrap instead of running off the page.
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,breakanywhere,commandchars=\\\{\}}

% Tables carry file paths and identifiers. Give them a size down so they fit.
% Deliberately NOT hyphenating monospace: a hyphen inserted into a path or a
% CSV header reads as part of the text.
\AtBeginEnvironment{longtable}{\footnotesize}
\AtBeginEnvironment{quote}{\small}

% Permit a break after the punctuation that appears inside those runs, so
% `arm,rep,op,...` and `a/long/path.sh` wrap at a sensible point.

  \@tfor\next:=#1\do{\ifx\next,\do,\else\ifx\next_\do_\else\ifx\next/\do/\else\next\fi\fi\fi}}}
\makeatother
